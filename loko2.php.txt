    <script>
        // Theme Engine Configuration Setup
        function initTheme() {
            const savedTheme = localStorage.getItem('loko_monitor_theme') || 'dark';
            document.documentElement.setAttribute('data-theme', savedTheme);
        }

        function toggleTheme() {
            const currentTheme = document.documentElement.getAttribute('data-theme');
            const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
            document.documentElement.setAttribute('data-theme', newTheme);
            localStorage.setItem('loko_monitor_theme', newTheme);
        }

        // Asynchronous Log Row Deletion Handler
        function deleteLog(logId) {
            if (confirm("Sigurado ka ba na gusto mo burahin ang log na ito, lods?")) {
                fetch('logger.php', {
                    method: 'DELETE',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ id: logId })
                })
                .then(response => response.json())
                .then(data => {
                    if (data.status === 'success') {
                        // Remove element tracking animation smoothly
                        const card = document.getElementById(`card-${logId}`);
                        if(card) {
                            card.style.transition = "all 0.3s ease";
                            card.style.opacity = "0";
                            card.style.transform = "scale(0.9)";
                            setTimeout(() => {
                                card.remove();
                                // Show fallback message if all empty
                                if(document.querySelectorAll('.log-card').length === 0) {
                                    document.getElementById('logs-container').innerHTML = '<div class="no-logs">No user logs captured yet layout.</div>';
                                }
                            }, 300);
                        }
                    } else {
                        alert("Error: Hindi nabura ang log.");
                    }
                })
                .catch(err => {
                    alert("Network error processing log target removal.");
                });
            }
        }

        // Initialize display rules on startup
        initTheme();
    </script>
