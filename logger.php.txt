<?php
header('Content-Type: application/json');

$logFile = 'logs.json';

// Handle Action Deletions
if ($_SERVER['REQUEST_METHOD'] === 'DELETE') {
    $input = json_decode(file_get_contents('php://input'), true);
    if (isset($input['id']) && file_exists($logFile)) {
        $currentData = json_decode(file_get_contents($logFile), true) ?? [];
        
        // Filter out the item matching the unique id
        $filteredData = array_values(array_filter($currentData, function($item) use ($input) {
            return $item['id'] !== $input['id'];
        }));
        
        file_put_contents($logFile, json_encode($filteredData, JSON_PRETTY_PRINT));
        echo json_encode(['status' => 'success']);
        exit;
    }
}

// Get IP Address
$ip = $_SERVER['REMOTE_ADDR'];
if (!empty($_SERVER['HTTP_CLIENT_IP'])) {
    $ip = $_SERVER['HTTP_CLIENT_IP'];
} elseif (!empty($_SERVER['HTTP_X_FORWARDED_FOR'])) {
    $ip = $_SERVER['HTTP_X_FORWARDED_FOR'];
}

// Simple free IP location fallback
$location = "Unknown Location";
if($ip !== '127.0.0.1' && $ip !== '::1') {
    $geo = @json_decode(file_get_contents("http://ip-api.com/json/{$ip}"), true);
    if($geo && $geo['status'] == 'success') {
        $location = $geo['city'] . ', ' . $geo['country'];
    }
}

// Capture JS Payload
$input = json_decode(file_get_contents('php://input'), true);

if ($input) {
    $currentData = [];
    if (file_exists($logFile)) {
        $currentData = json_decode(file_get_contents($logFile), true) ?? [];
    }
    
    $newLog = [
        'id' => uniqid(),
        'date_time' => date('Y-m-d H:i:s'),
        'ip' => $ip,
        'location' => $location,
        'device' => $input['device'] ?? 'Unknown',
        'network_status' => $input['networkStatus'] ?? 'Unknown',
        'battery' => $input['battery'] ?? 'Unknown',
        'time_on_page' => $input['timeOnPage'] ?? '0s',
        'click_count' => $input['clickCount'] ?? 0,
        'clicked_buttons' => $input['clickedButtons'] ?? [],
        'session_status' => $input['sessionStatus'] ?? 'Active'
    ];
    
    array_unshift($currentData, $newLog);
    file_put_contents($logFile, json_encode($currentData, JSON_PRETTY_PRINT));
    echo json_encode(['status' => 'success']);
    exit;
}

echo json_encode(['status' => 'error', 'message' => 'No action processed']);
